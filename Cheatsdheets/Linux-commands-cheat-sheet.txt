 #linux commands cheat sheet

this cheat sheet contains commonly 50  used Linux commands

#1. Navigation commands

NO.  COMMAND             DESCRIPTION                                        EXAMPLES
1.    pwd               show current working directory                      pwd
2.    ls                 List file and directory                            ls
3.    ls -l             List files in long format                           ls -l
4.    ls -a             show hidden files                                   ls -a   
5.    ls -la            Long listing including hidden files                 is -la
6.    cd                 Change directory                                  cd Downloads
7.    cd ..              Go to parent directory                              cd ..
8.    cd ~               Go to home directory                                cd ~

#2. FILE AND DIRECTORY COMMANDS

NO. COMMAND              DESCRIPTION                                         EXAMPLES
9.  mkdir                create a directory                                mkdir project
10. rmdir                 remove an empty directory                        rmdir test
11. touch                 create an empty file                            touch note.txt
12. cp                   copy file or directories                         cp file1 file2
13. mv                   move or rename files                            mv old.txt new.txt
14. rm                    Delete a directory                             rm  note.txt
15. rm -r                 Delete a directory and its contents             rm -r project

#3. VIEW FILE CONTENTS

NO. COMMAND                DESCRIPTION                                       EXAMPLES
16. cat                     Display file contents                             cat notes.txt
17. less                    view a file page by page                         less log.txt
18. head                    show the first 10 lines                          head notes.txt
19. tail                   show the last 10 lines                           tail notes.txt
20. grep                   search for text in a file                        grep linux note.txt

#4. SEARCH COMMANDS

NO. COMMAND                   DESCRIPTION                                     EXAMPLES
21. find                        search for files                           find ~ -name notes.txt
22. which                      show where a command is located                  which ls
23. whereis                  locate command, source and manual                  whereis bash

#5 FILE PERMISSION 

NO.  COMMAND                   DESCRIPTION                                        EXAMPLES
24. chmod                     Change file permissions                           chmod 755 script.sh
25.  chown                     Change file owner                                sudo chown hp note.txt
26. Chgrp                      Change file group                                sudo chgrp developers file.txt

#6. USER MANAGEMENT

NO.  COMMAND                      DESCRIPTION                                    EXAMPLES
27.  whoami                      Display current user                           whoami
28.  id                          Display user and group ids                       id
29.   useradd                    Create a new user                                sudo useradd john 
30.  passwd                     set or change a user's password                  sudo passwd john
31.  usermod                    modify a user account                             sudo usermod -aG sudo john

#7. PROCESS MANAGEMENT

NO. COMMAND                      DESCRIPTION                                       EXAMPLES
31. ps                            Show running processes                              ps aux
32. top                           Display running processes in real time            top
33. htop                         interactive process viewer                        htop
34. kill                        terminate a process                                 kill 1234
35. kill all                     kill processes by name                            killall firefox

#8. SERVICE MANAGEMENT
NO. COMMAND                        DESCRIPTION                                      EXAMPLES
36. systemctl status             check service status                              systemctl status ssh
37. systemctl start              start a service                                   sudo systemctl start nginx 
38. systemctl stop               stop a service                                    sudo systemctl stop nginx
39. systemctl restart            restart a service                                 sudo systemctl restart nginx
40. systemctl enable             enable a service                                  sudo systemctl enable nginx

#9. NETWORKING COMMANDS
NO. COMMAND                        DESCRIPTION                                       EXAMPLES
41. ip a                          show ip addresses                                     ip a
42. ping                         test network connectivity                           ping google.com
43. curl                          transfer data from a url                           curl https://jw.org
44. ssh                           connect to a remote server                         ssh username@ip-address
45. hostname -l                   Display ip address                                 hostname -i

#10. PACKAGE MANAGEMENT
NO. COMMAND                       DESCRIPTION                                         EXAMPLES
46. apt update                    update package list                                sudo apt update
47. apt upgrade                   upgrade installed packages                         sudo apt upgrade
48. apt install                   install a package                                  sudo apt install nginx
49. apt remove                    remove a package                                   sudo apt remove nginx
50. apt autoremove                remove unused packages                             sudo apt autoremove
 
#11. SYSTEM INFORMATION
NO.  COMMAND                       DESCRIPTION                                           EXAMPLES
51.  uname -a                       Display system information                        uname -a
52.  hostname                       Display system hostname                           hostname
53.  date                           show current date and time                          date
54.  uptime                         show system uptime                                 uptime
